# `POST /base/init` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `POST /base/init` that provisions a LarkBase year-base with partition tables `itec_001..itec_250` (41 fields each), idempotent + resumable, credentials passed per-request via headers.

**Architecture:** Clean architecture on Express. A per-request `larkAuth` middleware exchanges header app credentials for a cached tenant token and attaches a request-scoped `LarkGateway` (Lark **base/v3** REST via `fetch`). The `ProvisionYearBase` use-case diffs desired partitions against what Lark already has (Lark = source of truth) and creates only what is missing, within a soft time budget.

**Tech Stack:** Node.js ≥20 (global `fetch`, `node:test`, `node:assert`), Express 4. No new npm deps.

## Global Constraints

- **Node ≥ 20**, ESM (`"type":"module"`), no new dependencies (MySQL deferred).
- **Partition count = 250**; table names `itec_001`..`itec_250` (zero-pad 3).
- **41 fields**, names verbatim from schema (incl. spaces `/` `%`); types per spec §3.
- **Credentials from headers only** (`X-Lark-App-Id`, `X-Lark-App-Secret`); **never log `app_secret`**; server stores no Lark creds in env.
- **Lark base domain:** `https://open.larksuite.com` (config `config.lark.baseDomain`).
- **Lark rate limit:** batch/method throttling returns Lark `code` (e.g. `1254291`/`800004135`-class) — retry with backoff.
- Every Lark response is an envelope `{ code, msg, data }`; `code === 0` = success.
- Test runner: `npm test` (`node --test`). One assertion focus per test.

```js
// src/domain/services/partition.js
export const PARTITION_COUNT = 250;

export function partitionName(n) {
  return `itec_${String(n).padStart(3, '0')}`;
}

const RE = /^itec_(\d{3})$/;
export function parsePartitionNo(name) {
  const m = RE.exec(name);
  if (!m) return null;
  return Number(m[1]);
}
```

```js
// src/infrastructure/lark/tokenCache.js
import { LarkAuthError } from '../../domain/errors.js';

export function createTokenCache({ baseDomain, fetchFn = fetch, now = Date.now }) {
  const store = new Map(); // appId -> { token, expiresAt }

  async function getToken(appId, appSecret) {
    const cached = store.get(appId);
    if (cached && cached.expiresAt > now()) return cached.token;

    const res = await fetchFn(`${baseDomain}/open-apis/auth/v3/tenant_access_token/internal`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ app_id: appId, app_secret: appSecret }),
    });
    const body = await res.json();
    if (body.code !== 0) throw new LarkAuthError(body.msg || `token exchange failed (code ${body.code})`);

    const expiresAt = now() + (body.expire - 60) * 1000; // refresh 60s early
    store.set(appId, { token: body.tenant_access_token, expiresAt });
    return body.tenant_access_token;
  }

  return { getToken };
}
```

```js
// src/infrastructure/lark/LarkGatewayHttp.js
import { BaseNotFoundError } from '../../domain/errors.js';

export function createLarkGateway({ token, baseDomain, fetchFn = fetch }) {
  const auth = { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' };
  const call = async (path, opts = {}) => {
    const res = await fetchFn(`${baseDomain}${path}`, { ...opts, headers: auth });
    return res.json();
  };

  async function getBase(baseId) {
    const b = await call(`/open-apis/base/v3/bases/${baseId}`);
    if (b.code !== 0) throw new BaseNotFoundError(baseId);
  }

  async function listTables(baseId) {
    const out = [];
    let pageToken;
    do {
      const qs = new URLSearchParams({ page_size: '100', ...(pageToken ? { page_token: pageToken } : {}) });
      const b = await call(`/open-apis/base/v3/bases/${baseId}/tables?${qs}`);
      if (b.code !== 0) throw new BaseNotFoundError(baseId);
      for (const it of b.data.items ?? []) out.push({ tableId: it.table_id ?? it.id, name: it.name });
      pageToken = b.data.has_more ? b.data.page_token : undefined;
    } while (pageToken);
    return out;
  }

  async function createTable(baseId, name, fields) {
    const b = await call(`/open-apis/base/v3/bases/${baseId}/tables`, {
      method: 'POST', body: JSON.stringify({ table: { name, fields } }),
    });
    if (b.code !== 0) { const e = new Error(`createTable failed: ${b.msg}`); e.larkCode = b.code; throw e; }
    return b.data.table_id ?? b.data.table?.id;
  }

  async function listFields(baseId, tableId) {
    const names = [];
    let pageToken;
    do {
      const qs = new URLSearchParams({ page_size: '100', ...(pageToken ? { page_token: pageToken } : {}) });
      const b = await call(`/open-apis/base/v3/bases/${baseId}/tables/${tableId}/fields?${qs}`);
      if (b.code !== 0) { const e = new Error(`listFields failed: ${b.msg}`); e.larkCode = b.code; throw e; }
      for (const it of b.data.items ?? []) names.push(it.name ?? it.field_name);
      pageToken = b.data.has_more ? b.data.page_token : undefined;
    } while (pageToken);
    return names;
  }

  async function createField(baseId, tableId, field) {
    const b = await call(`/open-apis/base/v3/bases/${baseId}/tables/${tableId}/fields`, {
      method: 'POST', body: JSON.stringify(field),
    });
    if (b.code !== 0) { const e = new Error(`createField failed: ${b.msg}`); e.larkCode = b.code; throw e; }
  }

  return { getBase, listTables, createTable, listFields, createField };
}
```

```js
// src/infrastructure/web/middlewares/larkAuth.js
import { LarkAuthError } from '../../../domain/errors.js';

export function larkAuth({ tokenCache, createGateway, baseDomain }) {
  return async function larkAuthMiddleware(req, res, next) {
    const appId = req.headers['x-lark-app-id'];
    const appSecret = req.headers['x-lark-app-secret'];
    if (!appId || !appSecret) {
      return res.status(401).json({ error: 'invalid or missing Lark credentials' });
    }
    try {
      const token = await tokenCache.getToken(appId, appSecret); // never logged
      req.larkGateway = createGateway({ token, baseDomain });
      next();
    } catch (err) {
      if (err instanceof LarkAuthError) return res.status(401).json({ error: 'invalid or missing Lark credentials' });
      next(err);
    }
  };
}
```

```js
// src/application/use-cases/ProvisionYearBase.js
export class ProvisionYearBase {
  constructor({ fieldSchema, fieldNames, partitionCount, partitionName, parsePartitionNo, now = Date.now }) {
    this.fieldSchema = fieldSchema;
    this.fieldNames = fieldNames;
    this.partitionCount = partitionCount;
    this.partitionName = partitionName;
    this.parsePartitionNo = parsePartitionNo;
    this.now = now;
  }

  async execute({ gateway, base, year, budgetMs }) {
    await gateway.getBase(base); // throws BaseNotFoundError -> controller maps 404

    const tables = await gateway.listTables(base);
    const byNo = new Map();
    for (const t of tables) {
      const no = this.parsePartitionNo(t.name);
      if (no != null) byNo.set(no, t.tableId);
    }

    const start = this.now();
    const budgetLeft = () => this.now() - start < budgetMs;

    let existing = 0;
    let created = 0;
    let remaining = 0;
    let stopped = false;

    for (let n = 1; n <= this.partitionCount; n++) {
      const tableId = byNo.get(n);

      if (tableId) {
        existing++;
        const have = new Set(await gateway.listFields(base, tableId));
        const missing = this.fieldSchema.filter((f) => !have.has(f.name));
        if (missing.length === 0) continue;              // partition complete
        if (stopped || !budgetLeft()) { stopped = true; remaining++; continue; }
        for (const f of missing) await gateway.createField(base, tableId, f);
      } else {
        if (stopped || !budgetLeft()) { stopped = true; remaining++; continue; }
        await gateway.createTable(base, this.partitionName(n), this.fieldSchema);
        created++;
      }
    }

    return { year, base, total: this.partitionCount, existing, created, remaining, done: remaining === 0 };
  }
}
```

```js
import { BaseNotFoundError } from '../../../domain/errors.js';

const BUDGET_MS = 25000;

export class BaseController {
  constructor({ provisionYearBase, budgetMs = BUDGET_MS }) {
    this.provisionYearBase = provisionYearBase;
    this.budgetMs = budgetMs;
  }

  init = async (req, res, next) => {
    const { year, base } = req.body ?? {};
    if (!Number.isInteger(year) || year < 2000 || year > 2100) {
      return res.status(400).json({ error: 'year must be an integer year (CE)' });
    }
    if (!base || typeof base !== 'string') {
      return res.status(400).json({ error: 'base is required' });

    try {
      const summary = await this.provisionYearBase.execute({
        gateway: req.larkGateway, base, year, budgetMs: this.budgetMs,
      });
      res.status(200).json(summary);
    } catch (err) {
      if (err instanceof BaseNotFoundError) return res.status(404).json({ error: err.message });
      next(err);
    }
  };
```
